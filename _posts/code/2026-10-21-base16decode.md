/*

 *
 *    Description:  一个支持在编译时 decode hex (base16 编码) 的
 *                  head-only 工具
 *

 *                          然后，可以使用 s.c_str() 或 s[7] 获取其中的数据
 *
 *                  如果处于 C 语言环境，仅提供运行时 decode 工具
 *

#ifndef BALTAM_TOOLBOX_BASE16_H
#define BALTAM_TOOLBOX_BASE16_H

#ifdef __cplusplus
#include <algorithm>
#endif

// 将单个 hex 字符在编译时转为 raw data
// char2hex('F') 返回 0x0F
inline constexpr unsigned char char2hex(char c) {
    if (c >= '0' && c <= '9') {
        return c - '0';
    } else if (c >= 'A' && c <= 'F') {
        return c - 'A' + 10;
    } else if (c >= 'a' && c <= 'f') {
        return c - 'a' + 10;
    } else {
        return 0;
    }
}

// 将两个 hex 字符在编译时转为 raw data
// encode('7', 'B') 返回 0x7B
inline constexpr unsigned char decodeByte(char upper, char lower) {
    return (char2hex(upper) << 4) + char2hex(lower);
}

inline void decode(const char* const src, unsigned char* dest, int len) {
    for (int i = 0; i < len; i++)
        dest[i] = decodeByte(src[2 * i], src[2 * i + 1]);
}

#ifdef __cplusplus
namespace {
    // 一个支持 constexpr 的编译时数组
    template <class T, size_t N>
    class vector {
        T m_str[N]{};

    public:
        // 从单元素构造
        constexpr vector(const T c) noexcept { m_str[0] = c; }
        // 从 str 开始长度为 M 的数组构造
        constexpr vector(const T* str, size_t M) noexcept {
            for (size_t i = 0; i < std::min(N, M); ++i) m_str[i] = str[i];
        }
        // 从数组构造
        template <size_t M>
        constexpr vector(const T (&str)[M]) noexcept : vector{str, M} {}
        // 合并两个 vector
        template <size_t M>
        constexpr vector(const vector<T, M> v1,
                         const vector<T, N - M> v2) noexcept {
            for (size_t i = 0; i < M; ++i) m_str[i] = v1[i];
            for (size_t i = M; i < N; ++i) m_str[i] = v2[i - M];
        }
        // 取下标
        constexpr T operator[](size_t n) const noexcept {
            if (n >= N) return {};
            return m_str[n];
        }
        // 取长度
        constexpr size_t size(void) const noexcept { return N; }
        // 提取数组
        constexpr const T* c_str(void) const noexcept { return m_str; }
        // 取子串
        template <size_t M>
        constexpr vector<T, M> substr(size_t pos) const noexcept {
            return vector<T, M>(m_str + pos, M);
        }
    };
}  // namespace

// 将一个 hex vector 在编译时转为 raw data vector
template <size_t length>
constexpr ::vector<unsigned char, length> decodeVec(
    ::vector<char, 2 * length> coded_str) {
    if constexpr (length == 1) {
        return {decodeByte(coded_str[0], coded_str[1])};
    } else {
        constexpr size_t size_left = length / 2;
        auto left =
            decodeVec<size_left>(coded_str.template substr<size_left * 2>(0));

        constexpr size_t size_right = length - size_left;
        auto right = decodeVec<size_right>(
            coded_str.template substr<size_right * 2>(size_left * 2));
        return {left, right};
    }
}

// 将一个 hex 常量字符串在编译时转为 raw data vector
// 例如：constexpr auto s = decode<8>("0123456789ABCDEF");
// 然后，可以使用 s.c_str() 或 s[7] 获取其中的数据
template <size_t length>
constexpr ::vector<unsigned char, length> decode(
    const char (&coded_str)[2 * length + 1]) {
    return decodeVec<length>(coded_str);
}

#endif  // #ifdef __cplusplus

#endif  // #ifndef BALTAM_TOOLBOX_BASE16_H
